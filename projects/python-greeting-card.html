Animated Greeting Card

A creative Python project that displays warm pop-up messages in a heart-shaped pattern, with background music using Pygame.

Tools Used

Python, Tkinter, Pygame, Math, Threading

Main Skills

Animation timing, coordinate generation, audio playback, creative UI effects

Project Type

Creative coding / event-themed project

What the project does

The program generates heart-shaped points and opens small message windows around the screen while playing background music.

What it demonstrates

  • Using math formulas to create visual patterns
  • Combining Tkinter windows with Pygame audio
  • Using timing and threading to create animated effects
  • Creating a more emotional and creative coding output

import tkinter as tk
import random
import threading
import time
import math
import pygame

print("Success! Pygame version:", pygame.version.ver)

pygame.mixer.init()
SOUND = "银河快递 song.mp3"

def generate_heart_points(num_points):
    points = []
    for i in range(num_points):
        t = (i / num_points) * 2 * math.pi
        x = 16 * (math.sin(t) ** 3)
        y = -(13*math.cos(t) - 5 * math.cos(2 * t) - 2 * math.cos(3 * t) - math.cos(4 * t))
        points.append((x, y))
    return points

def show_warm_tip(x, y, screen_width, screen_height):
    window = tk.Tk()
    window_width, window_height = 200, 35
    
    scale = min(screen_width, screen_height) / 30
    pos_x = int(screen_width / 2 + x * scale - window_width / 2)
    
    vertical_offset = 100 
    pos_y = int(screen_height / 2 - vertical_offset + y * scale - window_height / 2)

    pos_x = max(0, min(pos_x, screen_width - window_width))
    pos_y = max(0, min(pos_y, screen_height - window_height))

    tips = [
        "Happy Mother's Day!", "Love You Always!", "Take a Break!",
        "You're beautiful!", "Thank You!", "Keep Smiling!", 
        "Stay Calm!", "You're the Best!", "Best Mom Ever!", 
        "Super Mom!", "You're Amazing!", "I LOVE YOU!!", 
        "My Hero, My Mom.", "Number 1 Mom.", "All my love."
    ]
    bg_color = random.choice([
        'lightsalmon', 'deeppink', 'mediumvioletred', 'orchid',
        'hotpink', 'palevioletred', 'lightcoral', 'salmon', 
        'darksalmon', 'red', 'violet', 'firebrick'
    ])

    label = tk.Label(window, text=random.choice(tips), bg=bg_color,
                     font=('Microsoft YaHei', 13, 'bold'),
                     width=25, height=2)
    label.pack()

    window.overrideredirect(True)
    window.attributes('-topmost', True)
    window.geometry(f"{window_width}x{window_height}+{pos_x}+{pos_y}")
