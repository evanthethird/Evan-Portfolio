What the project does

Players control a character to shoot bubbles at grumpy blobs. The game features a wave system, combo scoring, health bar, and collectible power-ups.

What it demonstrates

  • Building a complete game with HTML Canvas
  • Implementing game loops and physics
  • Creating AI enemy behavior and spawning
  • Adding particle effects and visual feedback
  • Managing game state and scoring systems

function update(dt) {
    if (game.mode !== "playing") return;

    // Wave system
    game.wave = 1 + Math.floor(game.elapsed / 18) + Math.floor(game.kills / 18);
    
    // Player movement (WASD / Arrow keys)
    let vx = 0, vy = 0;
    if (keys.w || keys.arrowup) vy -= 1;
    if (keys.s || keys.arrowdown) vy += 1;
    if (keys.a || keys.arrowleft) vx -= 1;
    if (keys.d || keys.arrowright) vx += 1;
    if (vx || vy) {
        const len = Math.hypot(vx, vy);
        player.x += (vx / len) * player.speed * dt;
        player.y += (vy / len) * player.speed * dt;
    }
    
    // Enemy AI - chase player
    for (let ei = enemies.length - 1; ei >= 0; ei--) {
        const e = enemies[ei];
        const angle = Math.atan2(player.y - e.y, player.x - e.x);
        e.x += Math.cos(angle) * e.speed * dt;
        e.y += Math.sin(angle) * e.speed * dt;
        
        // Collision with player
        if (Math.hypot(player.x - e.x, player.y - e.y) < player.r + e.r) {
            player.hp -= e.damage;
            // Damage feedback
        }
    }
    
    // Shooting
    if (mouse.down && player.fireCooldown <= 0) {
        const angle = Math.atan2(mouse.y - player.y, mouse.x - player.x);
        bullets.push({
            x: player.x + Math.cos(angle) * player.r,
            y: player.y + Math.sin(angle) * player.r,
            dx: Math.cos(angle) * 520,
            dy: Math.sin(angle) * 520,
            r: 7,
            life: 1.2
        });
        player.fireCooldown = 0.105;
    }
}
